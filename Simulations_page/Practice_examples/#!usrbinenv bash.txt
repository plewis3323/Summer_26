#!/usr/bin/env bash
# run_example.sh — one-shot driver for the capstone practice example
#
# Usage:
#   ./run_example.sh             # default 5000 events
#   ./run_example.sh 20000       # 20k events
#
# Assumes the course environment has been sourced.

set -euo pipefail

NEV=${1:-5000}

echo "===== [1/3] Pythia -> HepMC ($NEV events) ====="
./pythia_to_hepmc "$NEV" events.hepmc

echo
echo "===== [2/3] Geant4 calorimeter simulation ====="
./calo_sim run.mac

echo
echo "===== [3/3] ROOT analysis ====="
root -b -l -q analyze.C

echo
echo "Done. Plots in ./plots/  Tree in calo.root."
